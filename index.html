const SHEET_NAME = "工作表1"; const COACH_SHEET = "教練名單"; const PLAN_SHEET = "方案設定"; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME); if (data.action === "add") { const lastRow = sheet.getLastRow() + 1; sheet.getRange(lastRow, 1).setValue(data.id); sheet.getRange(lastRow, 2).setValue(data.coach); sheet.getRange(lastRow, 3).setValue(data.date).setNumberFormat("@"); sheet.getRange(lastRow, 4).setValue(data.start).setNumberFormat("@"); sheet.getRange(lastRow, 5).setValue(data.end).setNumberFormat("@"); sheet.getRange(lastRow, 6).setValue(data.type); sheet.getRange(lastRow, 7).setValue(data.status); sheet.getRange(lastRow, 8).setValue(data.createdAt); sheet.getRange(lastRow, 9).setValue("新增"); } if (data.action === "cancel") { const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(data.id)) { sheet.getRange(i + 1, 7).setValue("cancelled"); sheet.getRange(i + 1, 9).setValue("取消"); break; } } } if (data.action === "saveCoaches") { let cs = ss.getSheetByName(COACH_SHEET); if (!cs) cs = ss.insertSheet(COACH_SHEET); cs.clearContents(); cs.getRange(1, 1).setValue("名字"); data.coaches.forEach((c, i) => cs.getRange(i + 2, 1).setValue(c)); } if (data.action === "savePlans") { let ps = ss.getSheetByName(PLAN_SHEET); if (!ps) ps = ss.insertSheet(PLAN_SHEET); ps.clearContents(); ps.getRange(1, 1).setValue("教練"); ps.getRange(1, 2).setValue("方案"); ps.getRange(1, 3).setValue("已購買"); ps.getRange(1, 4).setValue("手動剩餘"); const entries = Object.entries(data.plans); entries.forEach(([coach, plan], i) => { ps.getRange(i + 2, 1).setValue(coach); ps.getRange(i + 2, 2).setValue(plan.type || ""); ps.getRange(i + 2, 3).setValue(plan.purchased || 0); ps.getRange(i + 2, 4).setValue(plan.manualRemaining ?? ""); }); } return ContentService .createTextOutput(JSON.stringify({ result: "ok" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ result: "error", message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); // 只抓最近兩個月的預約資料 const now = new Date(); const thisMonth = now.getFullYear() + "-" + String(now.getMonth() + 1).padStart(2, "0"); const lastMonthDate = new Date(now.getFullYear(), now.getMonth() - 1, 1); const lastMonth = lastMonthDate.getFullYear() + "-" + String(lastMonthDate.getMonth() + 1).padStart(2, "0"); const sheet = ss.getSheetByName(SHEET_NAME); const rows = sheet.getDataRange().getValues(); const bookings = []; for (let i = 1; i < rows.length; i++) { if (!rows[i][0]) continue; const dateStr = String(rows[i][2]); const month = dateStr.slice(0, 7); if (month !== thisMonth && month !== lastMonth) continue; bookings.push({ id: String(rows[i][0]), coach: String(rows[i][1]), date: String(rows[i][2]), start: String(rows[i][3]), end: String(rows[i][4]), type: String(rows[i][5]), status: String(rows[i][6]), createdAt: String(rows[i][7]), }); } // 教練名單 let coaches = []; const cs = ss.getSheetByName(COACH_SHEET); if (cs) { const crow = cs.getDataRange().getValues(); for (let i = 1; i < crow.length; i++) { if (crow[i][0]) coaches.push(String(crow[i][0])); } } // 方案設定 let plans = {}; const ps = ss.getSheetByName(PLAN_SHEET); if (ps) { const prow = ps.getDataRange().getValues(); for (let i = 1; i < prow.length; i++) { if (prow[i][0]) { plans[String(prow[i][0])] = { type: String(prow[i][1]), purchased: Number(prow[i][2]) || 0, manualRemaining: prow[i][3] !== "" ? Number(prow[i][3]) : null, }; } } } return ContentService .createTextOutput(JSON.stringify({ result: "ok", bookings, coaches, plans })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ result: "error", message: err.message })) .setMimeType(ContentService.MimeType.JSON); } }
